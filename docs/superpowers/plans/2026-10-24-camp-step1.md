```js
// 暖暖营地：一张 3072×4096 的竖画，宠物和装饰都用画的像素坐标；草地是一整块多边形，禁区是小溪、水塘、木台、篝火、树干
// 云函数和小程序共用，纯函数，有单测。画换了就按新画重新量这里的数字（scripts/camp/blockout.py 会照这里画白模）

const W = 3072
const H = 4096

function circle(cx, cy, r, n = 12) {
  const pts = []
  for (let i = 0; i < n; i++) {
    const a = (Math.PI * 2 * i) / n
    pts.push([Math.round(cx + r * Math.cos(a)), Math.round(cy + r * Math.sin(a))])
  }
  return pts
}

function ellipse(cx, cy, rx, ry, n = 16) {
  const pts = []
  for (let i = 0; i < n; i++) {
    const a = (Math.PI * 2 * i) / n
    pts.push([Math.round(cx + rx * Math.cos(a)), Math.round(cy + ry * Math.sin(a))])
  }
  return pts
}

const FIRE = [1536, 2500]

const SCENE = {
  width: W,
  height: H,
  // 能走能放的草地：上面是树林边缘，两边是灌木和石崖，下面是画框
  grass: [[520, 700], [2560, 700], [2960, 1500], [2990, 3950], [80, 3950], [110, 1500]],
  // 不能走也不能放的
  nogo: {
    stream: [[110, 1500], [340, 1500], [430, 2300], [400, 2900], [560, 3300], [470, 3600], [300, 3950], [80, 3950]],
    pond: ellipse(560, 3600, 320, 200),
    deck: [[1000, 2850], [1520, 2850], [1520, 3250], [1000, 3250]],
    fire: circle(FIRE[0], FIRE[1], 220),
    trees: [circle(700, 860, 90), circle(2500, 820, 90), circle(2820, 2300, 90), circle(2760, 3500, 90), circle(1150, 3700, 90)],
  },
  fire: FIRE,
  // 傍晚围着篝火坐的 12 个点
  seats: ellipse(FIRE[0], FIRE[1], 560, 330, 12),
  bowls: [[1700, 3100], [1880, 3140]],
  deck: { spots: [[1130, 2980], [1260, 2920], [1390, 2990]] },
  // 树林边缘的入口：新住户从这里走进来，鹤鹤从这里飞走
  entries: [[900, 720], [1536, 720], [2200, 720]],
  // 树荫下睡觉的点（每棵树下面一点）
  shade: [[700, 1020], [2500, 980], [2820, 2460], [2760, 3660]],
  pond: { spots: [[500, 3520], [640, 3640]] },
}

// 亲密度：喂了几顿 → 几级（1～5）
const LEVELS = [0, 3, 8, 15, 25]

// 远近：画面最下面 1，最上面 0.82
function scaleAt(y) {
  return 0.82 + (0.18 * y) / H
}

function pointInPoly(x, y, poly) {
  let inside = false
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    const [xi, yi] = poly[i]
    const [xj, yj] = poly[j]
    if (yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi) inside = !inside
  }
  return inside
}

function cross(o, a, b) {
  return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
}

function segmentsCross(a, b, c, d) {
  const d1 = cross(c, d, a)
  const d2 = cross(c, d, b)
  const d3 = cross(a, b, c)
  const d4 = cross(a, b, d)
  return ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))
}

function rectCorners(rect) {
  return [[rect.left, rect.top], [rect.right, rect.top], [rect.right, rect.bottom], [rect.left, rect.bottom]]
}

// 线段 a-b 有没有穿过多边形（端点在里面也算）
function segmentHitsPoly(a, b, poly) {
  if (pointInPoly(a[0], a[1], poly) || pointInPoly(b[0], b[1], poly)) return true
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    if (segmentsCross(a, b, poly[j], poly[i])) return true
  }
  return false
}

// 多边形和矩形有没有碰到：角在对方里面、或边相交
function polyHitsRect(poly, rect) {
  const corners = rectCorners(rect)
  if (corners.some(([x, y]) => pointInPoly(x, y, poly))) return true
  if (poly.some(([x, y]) => x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom)) return true
  for (let i = 0; i < 4; i++) {
    if (segmentHitsPoly(corners[i], corners[(i + 1) % 4], poly)) return true
  }
  return false
}

// 矩形完全在多边形里：四个角和中心都在里面，而且边不穿出去
function rectInPoly(rect, poly) {
  const corners = rectCorners(rect)
  const cx = (rect.left + rect.right) / 2
  const cy = (rect.top + rect.bottom) / 2
  if (!corners.every(([x, y]) => pointInPoly(x, y, poly)) || !pointInPoly(cx, cy, poly)) return false
  for (let i = 0; i < 4; i++) {
    const a = corners[i]
    const b = corners[(i + 1) % 4]
    for (let k = 0, j = poly.length - 1; k < poly.length; j = k++) {
      if (segmentsCross(a, b, poly[j], poly[k])) return false
    }
  }
  return true
}

function expand(rect, m) {
  return { left: rect.left - m, top: rect.top - m, right: rect.right + m, bottom: rect.bottom + m }
}

function nogoPolys(names) {
  const out = []
  ;(names || Object.keys(SCENE.nogo)).forEach((name) => {
    const v = SCENE.nogo[name]
    if (!v) return
    if (Array.isArray(v[0]) && Array.isArray(v[0][0])) v.forEach((p) => out.push(p))
    else out.push(v)
  })
  return out
}

// 矩形（外扩 margin）碰没碰到禁区；names 只查这几种
function hitsNogo(rect, margin = 0, names = null) {
  const r = expand(rect, margin)
  return nogoPolys(names).some((poly) => polyHitsRect(poly, r))
}

// 这一点能不能站：在草地里、不在任何禁区里
function inGrass(x, y) {
  if (!pointInPoly(x, y, SCENE.grass)) return false
  return !nogoPolys().some((poly) => pointInPoly(x, y, poly))
}

function levelOf(fed) {
  let level = 1
  LEVELS.forEach((need, i) => {
    if (fed >= need) level = i + 1
  })
  return level
}

// 下一级要喂到几顿；满级返回 null
function nextLevelAt(fed) {
  const next = LEVELS.find((need) => need > fed)
  return next === undefined ? null : next
}

// 早上的礼物：当天 5 点以后、今天还没领过
function giftDue(lastGift, today, hour) {
  return hour >= 5 && lastGift !== today
}

module.exports = {
  W, H, SCENE, LEVELS,
  circle, ellipse, scaleAt, pointInPoly, segmentHitsPoly, polyHitsRect, rectInPoly, expand, hitsNogo, inGrass,
  levelOf, nextLevelAt, giftDue,
}
```

```js
function foodById(id) {
  return FOODS.find((f) => f.id === id) || TREATS.find((t) => t.id === id) || null

function pickWeighted(entries, rand) {
  const total = entries.reduce((sum, e) => sum + e.weight, 0)
  let x = rand * total
  for (const e of entries) {
    if (x < e.weight) return e.value
    x -= e.weight
  }
  return entries[entries.length - 1].value
}

// 这一档里抽哪样：普通食物各 1，当季点心 SEASONAL_WEIGHT
function pickFood(rarity, termKey, rand) {
  const entries = FOODS.filter((f) => f.rarity === rarity).map((f) => ({ value: f.id, weight: 1 }))
    .concat(TREATS.filter((t) => t.rarity === rarity && t.term === termKey).map((t) => ({ value: t.id, weight: SEASONAL_WEIGHT })))
  return pickWeighted(entries, rand)
}

// 直接喂一口：爱吃 +3，不爱吃 +1，节气点心 +5；宠物或食物不存在 0
function feedGain(petId, foodId) {
  const pet = petById(petId)
  const food = foodById(foodId)
  if (!pet || !food) return 0
  if (food.term) return 5
  return pet.foods.indexOf(foodId) !== -1 ? 3 : 1
}

// 这份食物引谁来：爱吃它、食物的档和它一样、还没住进来的里随机一只；没有返回 null
function arrivalFor(foodId, residentIds, rand) {
  const food = foodById(foodId)
  if (!food || food.term) return null
  const candidates = PETS.filter((p) => p.foods.indexOf(foodId) !== -1 && p.rarity === food.rarity && residentIds.indexOf(p.id) === -1)
  if (!candidates.length) return null
  return candidates[Math.min(candidates.length - 1, Math.floor(rand * candidates.length))].id
}
```

```js
  const d = decorById(decorId)
  return { left: x - d.foot.w / 2, top: y - d.foot.h, right: x + d.foot.w / 2, bottom: y }
}

// 块的位置：anchor 对准落脚点
function patchRect(decorId, x, y) {
  const d = decorById(decorId)
  const left = x - d.anchor.x
  const top = y - d.anchor.y
  return { left, top, right: left + d.patch.w, bottom: top + d.patch.h }
}

const overlaps = (a, b) => a.left < b.right && b.left < a.right && a.top < b.bottom && b.top < a.bottom

// 这里为什么不能放（能放返回 ''）：云函数校验和页面提示用同一套话
function placeProblem(placed, decorId, x, y, ignoreUid) {
  const d = decorById(decorId)
  if (!d || typeof x !== 'number' || typeof y !== 'number' || !isFinite(x) || !isFinite(y)) return '放不到这里哦'
  const foot = footRect(decorId, x, y)
  if (!rectInPoly(foot, SCENE.grass)) return '只能放在草地上'
  if (hitsNogo(foot, 0)) return '这儿放不下，挪开一点'
  if (hitsNogo(foot, WATER_MARGIN, ['stream', 'pond'])) return '离水边远一点'
  const mine = expand(foot, OVERLAP_MARGIN)
  for (const p of placed) {
    if (p.uid === ignoreUid || !decorById(p.id)) continue
    if (overlaps(mine, expand(footRect(p.id, p.x, p.y), OVERLAP_MARGIN))) return '和别的东西挤着了'
  }
  return ''
}
```

```js
// gardens 文档（集合名没改）的完整形状：缺的字段补默认值；宅院 / 小岛 / 大地图时代的旧档迁成营地

const { WELCOME_BEANS } = require('./shared/garden')
const { OLD_TO_PET } = require('./shared/pets')

const LEGACY_FIELDS = ['estate', 'opened', 'inventory', 'placed', 'residents', 'collected', 'plots', 'unseen', 'currentSpecies', 'unlocked']

function pet(id, since) {
  return { id, since: since || '', fed: 0, fedToday: { date: '', n: 0 } }
}

// 旧档：暖豆按「打卡赚到的全部 + 新手礼」重算（花掉的全退），住户按原型搬进营地，其他全删
function adoptCamp(doc) {
  const beans = doc.beans || {}
  const welcomed = !!doc.welcomed
  const pets = []
  ;(doc.residents || []).forEach((r) => {
    const id = OLD_TO_PET[r.id]
    if (id && !pets.some((p) => p.id === id)) pets.push(pet(id, r.since))
  })
  if (welcomed && !pets.some((p) => p.id === 'doudou')) pets.push(pet('doudou', ''))
  return {
    camp: 1,
    welcomed,
    beans: { earned: beans.earned || {}, bonus: welcomed ? WELCOME_BEANS : 0, spent: 0 },
    gacha: doc.gacha,
    pets,
    grants: doc.grants,
    days: doc.days,
  }
}

function normalizeCamp(doc) {
  const src = doc && doc.camp ? { ...doc } : adoptCamp(doc || {})
  LEGACY_FIELDS.forEach((k) => delete src[k])
  const beans = src.beans || {}
  const gacha = src.gacha || {}
  const decor = src.decor || {}
  const two = (arr) => [arr && arr[0] ? arr[0] : null, arr && arr[1] ? arr[1] : null]
  return {
    ...src,
    camp: 1,
    welcomed: !!src.welcomed,
    beans: { earned: beans.earned || {}, bonus: beans.bonus || 0, spent: beans.spent || 0 },
    gacha: { pulls: gacha.pulls || 0, sincePity: gacha.sincePity || 0 },
    foods: src.foods || {},
    bowls: two(src.bowls),
    peeking: two(src.peeking),
    pets: (src.pets || []).map((p) => ({ id: p.id, since: p.since || '', fed: p.fed || 0, fedToday: p.fedToday || { date: '', n: 0 } })),
    arrivals: src.arrivals || [],
    decor: { owned: decor.owned || {}, placed: decor.placed || [] },
    seq: src.seq || 0,
    gift: src.gift || '',
    tasted: src.tasted || [],
    grants: src.grants || [],
    days: src.days || {},
  }
}

// 第一次进来、或旧档刚迁完，要存一次
function migrated(doc, camp) {
  return !doc || doc.camp !== camp.camp
}

module.exports = { normalizeCamp, migrated }
```

```js
const { AppError, CODES } = require('../lib/errors')
const { normalizeCamp, migrated } = require('../lib/camp-state')
const { dateKey } = require('../lib/shared/date')
const { RARITIES, SPIN_COST, WELCOME_BEANS, PITY, beanBalance, rollRarity, currentTerm } = require('../lib/shared/garden')
const { PETS, petById } = require('../lib/shared/pets')
const { FOODS, TREATS, foodById, pickFood, feedGain, arrivalFor } = require('../lib/shared/foods')
const { DECOR, GOLD_TENT, decorById, placeProblem } = require('../lib/shared/decor')
const { levelOf, nextLevelAt, giftDue } = require('../lib/shared/camp')

const HIGH_RARITIES = ['epic', 'legendary']
const STARTER_PET = 'doudou'
const FEEDS_PER_DAY = 3
const GIFT_BEANS = 5
const GIFT_BEANS_CHANCE = 0.7
const GIFT_LEVEL = 4
// 没来的宠物在图册里的提示：按它爱吃的第一样食物的口味
const TASTE_HINT = {
  xiaomi: '它好像爱吃香香的小粒粮', huluobo: '它好像爱吃脆脆的', lianzi: '它好像爱吃清甜的', shizi: '它好像爱吃软软甜甜的',
  xiaoyugan: '它闻到鱼味就来', xigua: '它好像爱吃多汁的', kaohongshu: '它好像爱吃热乎乎的', hetao: '它好像爱嗑坚果',
  juzi: '它好像爱吃酸酸甜甜的', zhusun: '它好像爱吃竹子', chaye: '它好像爱喝茶', shanzha: '它只在清晨出现，爱吃红红的小果子',
}

// hourOf：本地钟点（礼物 5 点后才有；云函数默认按北京时间算）
function createCampService({ repo, now, random = Math.random, getGrant = () => null, forest = null, hourOf = null }) {
  const today = () => dateKey(now())
  const hour = () => (hourOf ? hourOf() : (now().getUTCHours() + 8) % 24)

  async function load(openid) {
    return normalizeCamp(await repo.getGarden(openid))
  }

  function termInfo() {
    const term = currentTerm(today())
    if (!term) return null
    return { name: term.name, treats: TREATS.filter((t) => t.term === term.key).map((t) => t.id) }
  }

  const residentIds = (camp) => camp.pets.map((p) => p.id)
  const isFull = (camp) => PETS.every((p) => residentIds(camp).indexOf(p.id) !== -1)
  const giftReady = (camp) => giftDue(camp.gift, today(), hour()) && camp.pets.some((p) => levelOf(p.fed) >= GIFT_LEVEL)

  function unplaced(camp) {
    const placedCount = {}
    camp.decor.placed.forEach((p) => {
      placedCount[p.id] = (placedCount[p.id] || 0) + 1
    })
    return DECOR.filter((d) => (camp.decor.owned[d.id] || 0) - (placedCount[d.id] || 0) > 0)
      .map((d) => ({ id: d.id, count: (camp.decor.owned[d.id] || 0) - (placedCount[d.id] || 0) }))
  }

  function view(camp, extra = {}) {
    return {
      beans: Math.max(0, beanBalance(camp)),
      spinCost: SPIN_COST,
      pityLeft: PITY - camp.gacha.sincePity,
      rates: RARITIES.map(({ id, label, weight }) => ({ id, label, weight })),
      term: termInfo(),
      foods: FOODS.concat(TREATS).filter((f) => camp.foods[f.id] > 0).map((f) => ({ id: f.id, count: camp.foods[f.id] })),
      bowls: [0, 1].map((i) => ({ food: camp.bowls[i], peeking: camp.peeking[i] })),
      pets: camp.pets.map((p) => ({ id: p.id, since: p.since, fed: p.fed, level: levelOf(p.fed), next: nextLevelAt(p.fed) })),
      arrivals: camp.arrivals,
      decor: { owned: camp.decor.owned, placed: camp.decor.placed, unplaced: unplaced(camp) },
      giftReady: giftReady(camp),
      full: isFull(camp),
      ...extra,
    }
  }

  function takeFromBag(camp, foodId) {
    if (!foodById(foodId) || !(camp.foods[foodId] > 0)) throw new AppError(CODES.BAD_REQUEST, '背包里没有这个了')
    const foods = { ...camp.foods, [foodId]: camp.foods[foodId] - 1 }
    if (!foods[foodId]) delete foods[foodId]
    camp.foods = foods
  }

  function addToBag(camp, foodId) {
    camp.foods = { ...camp.foods, [foodId]: (camp.foods[foodId] || 0) + 1 }
  }

  function bowlIndex(input) {
    const bowl = (input || {}).bowl
    if (bowl !== 0 && bowl !== 1) throw new AppError(CODES.BAD_REQUEST, '没有这个食盆')
    return bowl
  }

  function findPlaced(camp, uid) {
    const placed = camp.decor.placed.find((p) => p.uid === uid)
    if (!placed) throw new AppError(CODES.NOT_FOUND, '找不到这件东西了')
    return placed
  }

  function checkSpot(camp, decorId, x, y, ignoreUid) {
    const problem = placeProblem(camp.decor.placed, decorId, x, y, ignoreUid)
    if (problem) throw new AppError(CODES.BAD_REQUEST, problem)
  }

  // 全员吃一顿（一天最多 FEEDS_PER_DAY 次）
  function feedAll(camp) {
    const date = today()
    camp.pets = camp.pets.map((p) => {
      const t = p.fedToday.date === date ? p.fedToday : { date, n: 0 }
      if (t.n >= FEEDS_PER_DAY) return { ...p, fedToday: t }
      return { ...p, fed: p.fed + 1, fedToday: { date, n: t.n + 1 } }
    })
  }

  return {
    async get(openid, input) {
      const doc = await repo.getGarden(openid)
      const camp = normalizeCamp(doc)
      let dirty = migrated(doc, camp)
      let welcome = null
      if (!camp.welcomed) {
        camp.welcomed = true
        camp.beans = { ...camp.beans, bonus: camp.beans.bonus + WELCOME_BEANS }
        if (residentIds(camp).indexOf(STARTER_PET) === -1) camp.pets = camp.pets.concat({ id: STARTER_PET, since: today(), fed: 0, fedToday: { date: '', n: 0 } })
        welcome = { beans: WELCOME_BEANS, petId: STARTER_PET }
        dirty = true
      }
      const grant = getGrant()
      let testGift = 0
      if (grant && today() <= grant.until && camp.grants.indexOf(grant.id) === -1) {
        camp.grants = camp.grants.concat(grant.id)
        camp.beans = { ...camp.beans, bonus: camp.beans.bonus + grant.beans }
        testGift = grant.beans
        dirty = true
      }
      const result = view(camp, { welcome, testGift })
      if (input && input.markSeen && camp.arrivals.length) {
        camp.arrivals = []
        dirty = true
      }
      if (dirty) await repo.saveGarden(openid, camp)
      result.hasProfile = !!(await repo.getProfile(openid))
      result.daily = null
      if (forest) result.daily = await forest.today(openid, today(), await repo.listMeals(openid, today()))
      return result
    },

    async spin(openid) {
      const camp = await load(openid)
      if (beanBalance(camp) < SPIN_COST) throw new AppError(CODES.NO_BEANS, '暖豆不够啦，再打卡一顿就有了')
      const term = currentTerm(today())
      const rarity = rollRarity(random(), camp.gacha.sincePity)
      const foodId = pickFood(rarity, term && term.key, random())
      const isNew = camp.tasted.indexOf(foodId) === -1
      camp.beans = { ...camp.beans, spent: camp.beans.spent + SPIN_COST }
      addToBag(camp, foodId)
      if (isNew) camp.tasted = camp.tasted.concat(foodId)
      camp.gacha = { pulls: camp.gacha.pulls + 1, sincePity: HIGH_RARITIES.indexOf(rarity) !== -1 ? 0 : camp.gacha.sincePity + 1 }
      await repo.saveGarden(openid, camp)
      return { foodId, rarity, isNew, beans: Math.max(0, beanBalance(camp)), pityLeft: PITY - camp.gacha.sincePity }
    },

    // 放进食盆：这一刻就定下来引谁（张望的就是它）
    async putFood(openid, input) {
      const bowl = bowlIndex(input)
      const camp = await load(openid)
      if (camp.bowls[bowl]) throw new AppError(CODES.BAD_REQUEST, '这个食盆已经有吃的了')
      const foodId = (input || {}).foodId
      takeFromBag(camp, foodId)
      camp.bowls = camp.bowls.map((f, i) => (i === bowl ? foodId : f))
      camp.peeking = camp.peeking.map((p, i) => (i === bowl ? arrivalFor(foodId, residentIds(camp), random()) : p))
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    async takeFood(openid, input) {
      const bowl = bowlIndex(input)
      const camp = await load(openid)
      const foodId = camp.bowls[bowl]
      if (!foodId) throw new AppError(CODES.BAD_REQUEST, '食盆是空的')
      addToBag(camp, foodId)
      camp.bowls = camp.bowls.map((f, i) => (i === bowl ? null : f))
      camp.peeking = camp.peeking.map((p, i) => (i === bowl ? null : p))
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    // 直接喂一只：爱吃 +3、不爱吃 +1、点心 +5，不限次
    async feed(openid, input) {
      const { petId, foodId } = input || {}
      const camp = await load(openid)
      const pet = camp.pets.find((p) => p.id === petId)
      if (!pet) throw new AppError(CODES.NOT_FOUND, '它还没住进来')
      takeFromBag(camp, foodId)
      const gain = feedGain(petId, foodId)
      camp.pets = camp.pets.map((p) => (p.id === petId ? { ...p, fed: p.fed + gain } : p))
      await repo.saveGarden(openid, camp)
      return view(camp, { gain, liked: gain >= 3, petId, foodId })
    },

    async buy(openid, input) {
      const decorId = (input || {}).decorId
      const d = decorById(decorId)
      if (!d || !d.price) throw new AppError(CODES.BAD_REQUEST, '这件商店里没有')
      const camp = await load(openid)
      const have = camp.decor.owned[decorId] || 0
      if (have >= d.max) throw new AppError(CODES.BAD_REQUEST, d.max === 1 ? '这件已经买过了' : '这件已经买够了')
      if (beanBalance(camp) < d.price) throw new AppError(CODES.NO_BEANS, `暖豆不够啦，${d.name}要 ${d.price} 暖豆，再攒攒`)
      camp.beans = { ...camp.beans, spent: camp.beans.spent + d.price }
      camp.decor = { ...camp.decor, owned: { ...camp.decor.owned, [decorId]: have + 1 } }
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    async place(openid, input) {
      const { decorId, x, y } = input || {}
      const camp = await load(openid)
      if (!unplaced(camp).some((u) => u.id === decorId)) throw new AppError(CODES.BAD_REQUEST, '背包里没有这个了')
      checkSpot(camp, decorId, x, y)
      camp.seq += 1
      camp.decor = { ...camp.decor, placed: camp.decor.placed.concat({ uid: `d${camp.seq}`, id: decorId, x, y }) }
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    async move(openid, input) {
      const { uid, x, y } = input || {}
      const camp = await load(openid)
      const placed = findPlaced(camp, uid)
      checkSpot(camp, placed.id, x, y, uid)
      camp.decor = { ...camp.decor, placed: camp.decor.placed.map((p) => (p.uid === uid ? { ...p, x, y } : p)) }
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    async stash(openid, input) {
      const { uid } = input || {}
      const camp = await load(openid)
      findPlaced(camp, uid)
      camp.decor = { ...camp.decor, placed: camp.decor.placed.filter((p) => p.uid !== uid) }
      await repo.saveGarden(openid, camp)
      return view(camp)
    },

    // 早上的礼物：4 级以上的住户里随机一只叼来
    async gift(openid) {
      const camp = await load(openid)
      if (!giftReady(camp)) throw new AppError(CODES.BAD_REQUEST, '今天没有礼物啦')
      const givers = camp.pets.filter((p) => levelOf(p.fed) >= GIFT_LEVEL)
      const giver = givers[Math.min(givers.length - 1, Math.floor(random() * givers.length))]
      const gift = { petId: giver.id, beans: 0, foodId: null }
      if (random() < GIFT_BEANS_CHANCE) {
        gift.beans = GIFT_BEANS
        camp.beans = { ...camp.beans, bonus: camp.beans.bonus + GIFT_BEANS }
      } else {
        const commons = FOODS.filter((f) => f.rarity === 'common')
        gift.foodId = commons[Math.min(commons.length - 1, Math.floor(random() * commons.length))].id
        addToBag(camp, gift.foodId)
      }
      camp.gift = today()
      await repo.saveGarden(openid, camp)
      return view(camp, { gift })
    },

    async album(openid) {
      const camp = await load(openid)
      const pets = PETS.map((p) => {
        const mine = camp.pets.find((r) => r.id === p.id)
        if (!mine) return { id: p.id, name: p.name, home: false, hint: TASTE_HINT[p.foods[0]] || '' }
        const level = levelOf(mine.fed)
        return {
          id: p.id, name: p.name, home: true, hook: p.hook, mood: p.mood, foods: p.foods.map((id) => foodById(id).name),
          since: mine.since, fed: mine.fed, level, next: nextLevelAt(mine.fed),
          story: p.story.map((text, i) => ({ text, open: i < level })),
        }
      })
      return {
        pets,
        progress: { pets: camp.pets.length, total: PETS.length },
        full: isFull(camp),
        foods: FOODS.concat(TREATS).map((f) => ({ id: f.id, name: f.name, rarity: f.rarity, term: f.term || '', tip: f.tip, tasted: camp.tasted.indexOf(f.id) !== -1 })),
        decor: DECOR.filter((d) => d.price > 0).map((d) => ({ id: d.id, name: d.name, category: d.category, owned: (camp.decor.owned[d.id] || 0) > 0 })),
      }
    },

    // 正餐打卡后由 meal 服务调用：全员 +1；哪个食盆有张望的宠物，它就进来（一次一只）；没人可引的食物由爱吃它的住户吃掉
    async arrive(openid) {
      const camp = await load(openid)
      feedAll(camp)
      let result = null
      for (let i = 0; i < 2 && !result; i++) {
        const foodId = camp.bowls[i]
        if (!foodId) continue
        const food = foodById(foodId)
        const comer = camp.peeking[i]
        if (comer && !camp.pets.some((p) => p.id === comer)) {
          camp.pets = camp.pets.concat({ id: comer, since: today(), fed: 0, fedToday: { date: '', n: 0 } })
          camp.arrivals = camp.arrivals.concat(comer)
          result = { id: comer, name: petById(comer).name, foodId, foodName: food.name }
        } else {
          const eater = camp.pets.find((p) => petById(p.id).foods.indexOf(foodId) !== -1 || food.term)
          if (!eater) continue
          camp.pets = camp.pets.map((p) => (p.id === eater.id ? { ...p, fed: p.fed + feedGain(p.id, foodId) } : p))
          result = { id: null, ate: eater.id, foodId, foodName: food.name }
        }
        camp.bowls = camp.bowls.map((f, k) => (k === i ? null : f))
        camp.peeking = camp.peeking.map((p, k) => (k === i ? null : p))
      }
      if (isFull(camp) && !camp.decor.owned[GOLD_TENT]) camp.decor = { ...camp.decor, owned: { ...camp.decor.owned, [GOLD_TENT]: 1 } }
      await repo.saveGarden(openid, camp)
      return result
    },
  }
}

module.exports = { createCampService }
```

```python
"""按 shared/camp.js 的 SCENE 画营地白模：包里的占位底图（切 16 块）+ 给 Codex 的底稿和标注图

    python3 scripts/camp/blockout.py

输出：miniprogram/images/camp/base/t00.jpg…t33.jpg（1536×2048 切 4×4）、art/guide/camp-blockout.jpg（1152×1536，Codex 底稿）、
art/guide/camp-blockout-labels.jpg（带名字，给人看）。需要 Pillow。
"""
import json
import os
import subprocess

from PIL import Image, ImageDraw, ImageFont

REPO = os.path.normpath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..', '..'))
OUT_TILES = os.path.join(REPO, 'miniprogram', 'images', 'camp', 'base')
GUIDE = os.path.join(REPO, 'art', 'guide')
FULL_W = 1536          # 包里占位底图的宽（高按 3:4）
GUIDE_W = 1152

GRASS = (150, 205, 95)
GRASS_DARK = (128, 182, 80)
FOREST = (62, 118, 70)
MOUNTAIN = (170, 196, 205)
SKY = (218, 236, 244)
WATER = (120, 190, 235)
WOOD = (196, 150, 96)
FIRE = (236, 120, 60)
STONE = (170, 165, 150)
TRUNK = (120, 86, 56)
CANOPY = (88, 150, 78)
BOWL = (210, 200, 180)


def scene():
    js = "console.log(JSON.stringify(require('./cloudfunctions/api/lib/shared/camp').SCENE))"
    return json.loads(subprocess.check_output(['node', '-e', js], cwd=REPO))


def draw(width, labels=False):
    S = scene()
    k = width / S['width']
    height = round(S['height'] * k)
    img = Image.new('RGB', (width, height), SKY)
    d = ImageDraw.Draw(img)
    P = lambda pts: [(x * k, y * k) for x, y in pts]
    # 远山、树林边缘、草地
    d.polygon(P([[0, 0], [S['width'], 0], [S['width'], 520], [2400, 300], [1600, 420], [900, 280], [0, 480]]), fill=MOUNTAIN)
    d.rectangle([0, 420 * k, width, 760 * k], fill=FOREST)
    d.rectangle([0, 700 * k, width, height], fill=GRASS_DARK)
    d.polygon(P(S['grass']), fill=GRASS)
    # 小溪、水塘
    d.polygon(P(S['nogo']['stream']), fill=WATER)
    d.polygon(P(S['nogo']['pond']), fill=WATER)
    # 木台、篝火、食盆
    d.polygon(P(S['nogo']['deck']), fill=WOOD)
    fx, fy = S['fire']
    d.ellipse([(fx - 220) * k, (fy - 140) * k, (fx + 220) * k, (fy + 140) * k], fill=STONE)
    d.ellipse([(fx - 120) * k, (fy - 90) * k, (fx + 120) * k, (fy + 60) * k], fill=FIRE)
    for bx, by in S['bowls']:
        d.ellipse([(bx - 70) * k, (by - 45) * k, (bx + 70) * k, (by + 45) * k], fill=BOWL, outline=STONE, width=max(1, int(4 * k)))
    # 树：树干圆 + 树冠
    for tree in S['nogo']['trees']:
        cx = sum(p[0] for p in tree) / len(tree)
        cy = sum(p[1] for p in tree) / len(tree)
        d.rectangle([(cx - 40) * k, (cy - 60) * k, (cx + 40) * k, (cy + 80) * k], fill=TRUNK)
        d.ellipse([(cx - 260) * k, (cy - 520) * k, (cx + 260) * k, (cy - 40) * k], fill=CANOPY)
    # 溪边石头
    for sx, sy in [[300, 1700], [520, 2600], [250, 3200], [900, 3750]]:
        d.ellipse([(sx - 70) * k, (sy - 45) * k, (sx + 70) * k, (sy + 45) * k], fill=STONE)
    if labels:
        font = ImageFont.load_default()
        def tag(x, y, text):
            d.text((x * k + 4, y * k + 4), text, fill=(0, 0, 0), font=font)
        tag(fx, fy, 'fire 篝火')
        tag(1000, 2850, 'deck 木台')
        tag(S['bowls'][0][0], S['bowls'][0][1] - 120, 'bowls 食盆')
        tag(200, 2400, 'stream 小溪')
        tag(420, 3600, 'pond 水塘')
        for i, (ex, ey) in enumerate(S['entries']):
            tag(ex, ey, f'entry {i + 1}')
        for sx, sy in S['seats']:
            d.ellipse([sx * k - 6, sy * k - 6, sx * k + 6, sy * k + 6], fill=(255, 255, 255))
        for sx, sy in S['shade']:
            d.ellipse([sx * k - 6, sy * k - 6, sx * k + 6, sy * k + 6], fill=(60, 60, 60))
        d.line(P(S['grass'] + [S['grass'][0]]), fill=(255, 255, 255), width=3)
    return img


def main():
    os.makedirs(OUT_TILES, exist_ok=True)
    os.makedirs(GUIDE, exist_ok=True)
    full = draw(FULL_W)
    tw, th = full.width / 4, full.height / 4
    total = 0
    for row in range(4):
        for col in range(4):
            tile = full.crop((round(col * tw), round(row * th), round((col + 1) * tw), round((row + 1) * th)))
            path = os.path.join(OUT_TILES, f't{row}{col}.jpg')
            tile.save(path, quality=72, optimize=True)
            total += os.path.getsize(path)
    draw(GUIDE_W).save(os.path.join(GUIDE, 'camp-blockout.jpg'), quality=88)
    draw(GUIDE_W, labels=True).save(os.path.join(GUIDE, 'camp-blockout-labels.jpg'), quality=88)
    print(f'底图 {full.width}×{full.height}，16 块共 {total // 1024} KB；底稿和标注图已写入 art/guide/')


if __name__ == '__main__':
    main()
```

```python
def font(size):
    for path in ['/usr/share/fonts/truetype/wqy/wqy-microhei.ttc', '/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc', '/usr/share/fonts/truetype/noto/NotoSansCJK-Regular.ttc']:

def save(img, *parts):

def pet_image(p):
    old = STAND_IN.get(p['id'])
    src = os.path.join(OLD_ITEMS, f'{old}.png') if old else None
    if src and os.path.exists(src):
        img = Image.open(src).convert('RGBA')
        img.thumbnail((PET_PX, PET_PX), Image.LANCZOS)
        canvas = Image.new('RGBA', (PET_PX, PET_PX), (0, 0, 0, 0))
        canvas.paste(img, ((PET_PX - img.width) // 2, PET_PX - img.height), img)
        return canvas
    canvas = Image.new('RGBA', (PET_PX, PET_PX), (0, 0, 0, 0))
    d = ImageDraw.Draw(canvas)
    color = BLOB.get(p['id'], (200, 200, 200))
    d.ellipse([40, 70, 200, 230], fill=color)
    d.ellipse([70, 30, 170, 130], fill=color)
    d.ellipse([95, 60, 110, 75], fill=(40, 40, 40))
    d.ellipse([130, 60, 145, 75], fill=(40, 40, 40))
    d.text((85, 150), p['name'], fill=(255, 255, 255), font=font(28))
    return canvas


def face(img):
    box = img.getbbox() or (0, 0, img.width, img.height)
    head = img.crop((box[0], box[1], box[2], box[1] + (box[2] - box[0])))
    head.thumbnail((FACE_PX, FACE_PX), Image.LANCZOS)
    canvas = Image.new('RGBA', (FACE_PX, FACE_PX), (0, 0, 0, 0))
    canvas.paste(head, ((FACE_PX - head.width) // 2, (FACE_PX - head.height) // 2), head)
    return canvas


def decor_patch(dc):
    w, h = round(dc['patch']['w'] * PATCH_SCALE), round(dc['patch']['h'] * PATCH_SCALE)
    img = Image.new('RGBA', (w, h), GRASS + (255,))
    d = ImageDraw.Draw(img)
    fw, fh = round(dc['foot']['w'] * PATCH_SCALE), round(dc['foot']['h'] * PATCH_SCALE)
    ax, ay = round(dc['anchor']['x'] * PATCH_SCALE), round(dc['anchor']['y'] * PATCH_SCALE)
    color = CATEGORY_COLOR[dc['category']]
    d.rounded_rectangle([ax - fw // 2, ay - fh, ax + fw // 2, ay], radius=6, fill=color, outline=(90, 70, 50))
    d.ellipse([ax - fw // 2, ay - 4, ax + fw // 2, ay + 4], fill=(110, 150, 70))
    d.text((ax - fw // 2 + 4, ay - fh + 4), dc['name'], fill=(50, 40, 30), font=font(12))
    # 羽化边：alpha 从边缘 6px 处渐变到 0
    mask = Image.new('L', (w, h), 0)
    ImageDraw.Draw(mask).rectangle([6, 6, w - 7, h - 7], fill=255)
    mask = mask.filter(ImageFilter.GaussianBlur(4))
    img.putalpha(mask)
    return img


def food_icon(f):

    d = ImageDraw.Draw(img)

    d.ellipse([2, 2, FOOD_PX - 3, FOOD_PX - 3], fill=color, outline=(90, 80, 70), width=2)
    d.text((10, 18), f['name'][:2], fill=(40, 30, 20), font=font(22))
    return img


def main():
```

```js
// 营地画面的换算（纯函数，有单测）：画的像素 ↔ 屏幕像素、底图切块、装饰块、拖拽幽灵、背包 / 商店 / 食盆的展示数据、色调

const { W, H, scaleAt } = require('./shared/camp')
const { RARITIES } = require('./shared/garden')
const { petById } = require('./shared/pets')
const { foodById } = require('./shared/foods')
const { DECOR, CATEGORIES, decorById, footRect } = require('./shared/decor')
const { SCENE } = require('./shared/camp')

const Z_DECOR = 100
const MARGIN_RATIO = 0.6
const BOWL_PX = 150 // 食盆在画上的直径（像素）

const fmt = (n) => Math.round(n * 100) / 100

function campLayout(win) {
  const k = win.width / W
  const width = W * k
  const height = H * k
  const margin = win.width * MARGIN_RATIO
  return { W, H, k, width, height, margin, viewWidth: width + margin * 2, viewHeight: height + margin * 2 }
}

function toScreen(x, y, L) {
  return { x: x * L.k, y: y * L.k }
}

// 手指 → 画的像素：worldRect 是 #world 此刻的 boundingClientRect（含缩放）
function fromTouch(clientX, clientY, rect, L) {
  const scale = rect.width / L.width
  return { x: ((clientX - rect.left) / scale) / L.k, y: ((clientY - rect.top) / scale) / L.k }
}

function baseTiles(L, base) {
  const prefix = base || '/images/camp/base/'
  const tiles = []
  const tw = L.width / 4
  const th = L.height / 4
  for (let row = 0; row < 4; row++) {
    for (let col = 0; col < 4; col++) {
      tiles.push({ key: `t${row}${col}`, image: `${prefix}t${row}${col}.jpg`, left: fmt(col * tw), top: fmt(row * th), width: fmt(tw), height: fmt(th) })
    }
  }
  return tiles
}

const petImage = (id) => `/images/camp/pets/${id}.png`
const petFace = (id) => `/images/camp/pets/${id}-face.png`
const foodImage = (id) => `/images/camp/foods/${id}.png`
const decorImage = (id, base) => `${base || '/images/camp/decor/'}${id}.png`
const zOf = (y) => Z_DECOR + Math.round(y / 4)

function patchView(d, x, y, L, base) {
  const s = scaleAt(y)
  return {
    image: decorImage(d.id, base),
    left: fmt((x - d.anchor.x * s) * L.k),
    top: fmt((y - d.anchor.y * s) * L.k),
    width: fmt(d.patch.w * s * L.k),
    height: fmt(d.patch.h * s * L.k),
    z: zOf(y),
  }
}

function decorViews(placed, L, base) {
  return placed
    .map((p) => ({ p, d: decorById(p.id) }))
    .filter((x) => x.d)
    .map(({ p, d }) => ({ uid: p.uid, id: p.id, name: d.name, x: p.x, y: p.y, ...patchView(d, p.x, p.y, L, base) }))
    .sort((a, b) => a.z - b.z)
}

// 拖着走的幽灵：块 + 落脚框（绿能放、红不能）
function ghostView(decorId, x, y, L, ok) {
  const d = decorById(decorId)
  const foot = footRect(decorId, x, y)
  return {
    ...patchView(d, x, y, L, ''),
    ok: !!ok,
    foot: { left: fmt(foot.left * L.k), top: fmt(foot.top * L.k), width: fmt((foot.right - foot.left) * L.k), height: fmt((foot.bottom - foot.top) * L.k) },
  }
}

function rarityOf(id) {
  return RARITIES.find((r) => r.id === id) || RARITIES[0]
}

function foodBag(foods) {
  return (foods || [])
    .map(({ id, count }) => {
      const f = foodById(id)
      if (!f) return null
      const r = rarityOf(f.rarity)
      return { id, name: f.name, count, rarity: r.id, rarityLabel: r.label, rarityColor: r.color, image: foodImage(id), term: f.term || '' }
    })
    .filter(Boolean)
}

function decorBag(unplaced) {
  return (unplaced || [])
    .map(({ id, count }) => {
      const d = decorById(id)
      return d ? { id, name: d.name, count, image: decorImage(id), category: d.category } : null
    })
    .filter(Boolean)
}

function shopGroups(owned, beans) {
  return CATEGORIES.map((c) => ({
    id: c.id,
    label: c.label,
    items: DECOR.filter((d) => d.category === c.id && d.price > 0).map((d) => {
      const have = (owned && owned[d.id]) || 0
      return { id: d.id, name: d.name, price: d.price, image: decorImage(d.id), have, max: d.max, soldOut: have >= d.max, canAfford: beans >= d.price }
    }),
  }))
}

function bowlViews(bowls, L) {
  return SCENE.bowls.map(([bx, by], index) => {
    const b = (bowls && bowls[index]) || { food: null, peeking: null }
    const size = BOWL_PX * scaleAt(by) * L.k
    return {
      index,
      left: fmt(bx * L.k - size / 2),
      top: fmt(by * L.k - size / 2),
      size: fmt(size),
      z: zOf(by),
      food: b.food || '',
      foodImage: b.food ? foodImage(b.food) : '',
      peeking: b.peeking || '',
      peekImage: b.peeking ? petFace(b.peeking) : '',
    }
  })
}

// 一天的光：傍晚偏橙、夜里偏蓝
function tintFor(hour) {
  if (hour >= 19 || hour < 6) return 'night'
  if (hour >= 17) return 'dusk'
  return ''
}

// 打卡结果页的横幅
function arrivalCard(arrival) {
  if (!arrival) return null
  if (arrival.id) {
    const p = petById(arrival.id)
    return { image: petImage(arrival.id), text: `${p ? p.name : ''}闻着${arrival.foodName}的香味，搬进营地啦！` }
  }
  const p = petById(arrival.ate)
  return { image: petImage(arrival.ate), text: `${p ? p.name : ''}把食盆里的${arrival.foodName}吃了，营地里没有别的小动物爱吃它啦` }
}

// 让画上的点 (x, y) 停在屏幕正中：movable-view 的 x / y 是 view 左上角相对可视区的位移，缩放以 view 中心为原点
function focusOffset(x, y, L, win, scale) {
  const cx = L.viewWidth / 2
  const cy = L.viewHeight / 2
  const px = L.margin + x * L.k
  const py = L.margin + y * L.k
  return { x: win.width / 2 - cx - scale * (px - cx), y: win.height / 2 - cy - scale * (py - cy), scale }
}

module.exports = {
  Z_DECOR, campLayout, toScreen, fromTouch, baseTiles, decorViews, ghostView, zOf,
  petImage, petFace, foodImage, decorImage, rarityOf, foodBag, decorBag, shopGroups, bowlViews, tintFor, arrivalCard, focusOffset,
}
```

```
{ id, name, image, face, width, left, bottom, z, flip, action, cssAction, duration, hidden,

  bubble, bubbleWidth, bubbleLeft, bubbleBottom, emoji, tapping, shadow: { left, bottom, width, height } }
```

- `pets` 是 `view.pets`（`[{ id, level, fed }]`）；`decor` 是 `view.decor.placed`；坐标全是画的像素，页面按 `L.k` 缩放；`duration` 是这一步走多久（毫秒），页面用 `transition`

- 走动：目标点直线走，`duration = 距离 / speed × 1000`；路线穿过禁区就在垂线方向 ±360px 找一个绕行点，两段都不穿才走；找不到就不去

```js
    .filter((a) => a.pet)
    .map((a) => ({
      ...a, id: a.pet.id, x: SCENE.fire[0], y: SCENE.fire[1] + 700, right: true, action: 'idle', goal: null, duration: 0, hidden: false,
      bubble: '', emoji: '', tapping: false, sleepStyle: 'A', timer: null, bubbleTimer: null, tapTimer: null, holdGift: false,
    }))

  // ---------------------------------------------------------------- 地方

  function nogoPolys() {
    const n = SCENE.nogo
    return [n.stream, n.pond, n.deck, n.fire].concat(n.trees)
  }

  // 装饰上的点：{ x, y, action, uid, name }
  function decorSpots(action) {
    const out = []
    placed.forEach((p) => {
      const d = decorById(p.id)
      if (!d) return
      const rect = patchRect(p.id, p.x, p.y)
      const s = scaleAt(p.y)
      d.spots.forEach((sp) => {
        if (sp.action !== action) return
        out.push({ x: rect.left + sp.x * s, y: rect.top + sp.y * s, action, uid: p.uid, name: d.name })
      })
    })
    return out
  }

  function taken(spot, self) {
    return animals.some((a) => a !== self && a.goal && a.goal.spot && dist(a.goal.spot.x, a.goal.spot.y, spot.x, spot.y) < 1)
  }

  function free(spots, self) {
    return spots.filter((s) => !taken(s, self))
  }

  function sleepSpot(self) {
    const beds = free(decorSpots('sleep'), self)
    if (beds.length) return beds[0]
    const shade = free(SCENE.shade.map(([x, y]) => ({ x, y, action: 'sleep' })), self)
    if (shade.length) return shade[0]
    const deck = free(SCENE.deck.spots.map(([x, y]) => ({ x, y, action: 'sleep' })), self)
    return deck[0] || null
  }

  function awayFromOthers(x, y, self) {
    return animals.every((a) => a === self || a.hidden || dist(a.x, a.y, x, y) > self.pet.size * BODY_GAP)
  }

  // 草地上随机一点（离别人远、离中心近一些的优先）
  function randomGrass(self, tries = 30) {
    for (let i = 0; i < tries; i++) {
      const x = 200 + random() * (SCENE.width - 400)
      const y = 800 + random() * (SCENE.height - 1000)
      if (inGrass(x, y) && awayFromOthers(x, y, self)) return { x, y, action: 'idle' }
    }
    return null
  }

  function nearFire(self) {
    for (let i = 0; i < 30; i++) {
      const a = random() * Math.PI * 2
      const r = 320 + random() * 260
      const x = SCENE.fire[0] + r * Math.cos(a)
      const y = SCENE.fire[1] + r * Math.sin(a) * 0.6
      if (inGrass(x, y) && awayFromOthers(x, y, self)) return { x, y, action: 'idle' }
    }
    return null
  }

  // ---------------------------------------------------------------- 画面

  function sheetOf(a) {
    const s = sheets[a.id]
    if (!s || !animBase || bad.has(a.id)) return null
    const anim = frameFor(a.action, { sleepStyle: a.sleepStyle })
    return { sheet: `${animBase}${a.id}-${anim.sheet}.png`, anim: { ...anim, cols: COLS, rows: ROWS }, top: s.top }
  }

  function sprite(a) {
    const s = scaleAt(a.y)
    const width = a.pet.size * s * L.k
    const sh = sheetOf(a)
    const x = a.x * L.k
    const y = a.y * L.k
    const bubbleWidth = Math.min(BUBBLE_W * L.k, L.width)
    const bubbleX = Math.min(Math.max(x - bubbleWidth / 2, 0), L.width - bubbleWidth)
    const shadowW = width * SHADOW_RATIO
    return {
      id: a.id, name: a.pet.name, image: petImage(a.id), face: petFace(a.id), width, left: x - width / 2, bottom: L.height - y, z: zOf(a.y) + 1,
      flip: !a.right, action: a.action, cssAction: CSS_ACTIONS[a.action] || 'act-idle', duration: a.duration, hidden: a.hidden,
      sheet: sh ? sh.sheet : '', anim: sh ? sh.anim : null,
      bubble: a.bubble, bubbleWidth, bubbleLeft: bubbleX - (x - width / 2), bubbleBottom: width * (1 - (sh ? sh.top : 0.05)) + 4,
      emoji: a.action === 'sleep' ? '💤' : a.emoji, tapping: a.tapping,
      shadow: { left: x - shadowW / 2, bottom: L.height - y - shadowW / 6, width: shadowW, height: shadowW / 3 },
    }
  }

  function emit(a) {
    onChange(a.index, sprite(a))
  }

  function schedule(a, ms) {
    clock.clear(a.timer)
    a.timer = running ? clock.set(() => turn(a), ms) : null
  }

  function say(a, text, ms = BUBBLE_MS) {
    a.bubble = text
    clock.clear(a.bubbleTimer)
    a.bubbleTimer = clock.set(() => {
      a.bubble = ''
      emit(a)
    }, ms)
  }

  function heart(a) {
    a.tapping = true
    clock.clear(a.tapTimer)
    a.tapTimer = clock.set(() => {
      a.tapping = false
      emit(a)
    }, TAP_MS)
  }

  // ---------------------------------------------------------------- 走

  function detour(from, to) {
    const polys = nogoPolys()
    const blocked = (p, q) => polys.some((poly) => segmentHitsPoly([p.x, p.y], [q.x, q.y], poly))
    if (!blocked(from, to)) return [to]
    const dx = to.x - from.x
    const dy = to.y - from.y
    const len = Math.hypot(dx, dy) || 1
    const mid = { x: (from.x + to.x) / 2, y: (from.y + to.y) / 2 }
    for (const side of [1, -1, 2, -2]) {
      const via = { x: mid.x + ((-dy / len) * DETOUR) * side, y: mid.y + ((dx / len) * DETOUR) * side }
      if (inGrass(via.x, via.y) && !blocked(from, via) && !blocked(via, to)) return [via, to]
    }
    return null
  }

  function walkTo(a, spot, then) {
    const path = detour({ x: a.x, y: a.y }, spot)
    if (!path) return false
    a.goal = { spot, then, path }
    stepAlong(a)
    return true
  }

  function stepAlong(a) {
    const next = a.goal.path.shift()
    a.right = facingFor(next.x - a.x, a.right)
    a.duration = Math.max(300, Math.round((dist(a.x, a.y, next.x, next.y) / a.pet.speed) * 1000))
    a.x = next.x
    a.y = next.y
    a.action = a.pet.gait === 'fly' ? 'fly' : 'walk'
    emit(a)
    schedule(a, a.duration)
  }

  function arrived(a) {
    const g = a.goal
    a.duration = 0
    if (g.path.length) return stepAlong(a)
    const then = g.then
    a.goal = { spot: g.spot }
    then(a)
  }

  function doFor(a, action, ms, emoji = '') {
    a.action = action
    a.emoji = emoji
    emit(a)
    schedule(a, ms)
  }

  function rest(a, ms = between(REST_MS)) {
    a.goal = null
    a.emoji = ''
    doFor(a, 'idle', ms)
  }

  // ---------------------------------------------------------------- 一回合

  function away(a) {
    const h = hour()
    return !!a.pet.hours && (h < a.pet.hours[0] || h >= a.pet.hours[1])
  }

  function turn(a) {
    a.timer = null
    if (a.holdGift) return
    if (a.goal && a.goal.path) return arrived(a)
    if (away(a)) {
      a.hidden = true
      a.action = 'away'
      a.goal = null
      emit(a)
      return schedule(a, between(NIGHT_MS))
    }
    if (a.hidden) {
      const [ex, ey] = pick(SCENE.entries)
      a.hidden = false
      a.x = ex
      a.y = ey
    }
    const period = periodOf(hour())
    if (period === 'night') {
      if (a.action === 'sleep') return doFor(a, 'sleep', between(NIGHT_MS))
      const bed = sleepSpot(a)
      if (bed && walkTo(a, bed, (b) => { b.sleepStyle = b.level >= 2 && random() < 0.5 ? 'B' : 'A'; doFor(b, 'sleep', between(NIGHT_MS)) })) return
      return doFor(a, 'sleep', between(NIGHT_MS))
    }
    if (period === 'dusk') {
      const [sx, sy] = SCENE.seats[a.index % SCENE.seats.length]
      if (a.action === 'sit' && dist(a.x, a.y, sx, sy) < 1) return doFor(a, 'sit', between(DO_MS))
      if (walkTo(a, { x: sx, y: sy, action: 'sit' }, (b) => doFor(b, 'sit', between(DO_MS)))) return
      return rest(a)
    }
    if (a.action !== 'idle') return rest(a)
    if (period === 'morning') {
      const spot = nearFire(a)
      if (spot && walkTo(a, spot, (b) => rest(b, between(DO_MS)))) return
      return rest(a)
    }
    // 白天
    if (random() < PAIR_CHANCE && pair(a)) return
    const plays = free(decorSpots('play').concat(decorSpots('climb')), a)
    if (plays.length && random() < PLAY_CHANCE) {
      const spot = pick(plays)
      if (walkTo(a, spot, (b) => doFor(b, spot.action, between(DO_MS), '🎵'))) return
    }
    if (SUN_LOVERS.indexOf(a.id) !== -1 && random() < 0.5) {
      const sun = free(SCENE.deck.spots.map(([x, y]) => ({ x, y, action: 'sun' })), a)
      if (sun.length && walkTo(a, pick(sun), (b) => doFor(b, 'sun', between(NAP_MS), '☀️'))) return
    }
    if (SWIMMERS.indexOf(a.id) !== -1 && random() < 0.3) {
      const water = free(SCENE.pond.spots.map(([x, y]) => ({ x, y, action: 'soak' })), a)
      if (water.length && walkTo(a, pick(water), (b) => doFor(b, 'soak', between(DO_MS), '💧'))) return
    }
    if (random() < 0.15) return doFor(a, 'fidget', 1500)
    const spot = randomGrass(a)
    if (spot && walkTo(a, spot, (b) => rest(b))) return
    rest(a)
  }

  // 它们之间：a 走到 b 旁边，各说一句
  function pair(a) {
    const rule = PAIRS.find((r) => r.a === a.id)
    if (!rule) return false
    const b = animals.find((o) => o.id === rule.b && !o.hidden && o.action === 'idle' && !o.holdGift)
    if (!b) return false
    const spot = { x: b.x + (a.pet.size + b.pet.size) * 0.4 * (a.x < b.x ? -1 : 1), y: b.y + 20, action: 'idle' }
    if (!inGrass(spot.x, spot.y)) return false
    return walkTo(a, spot, (me) => {
      me.right = b.x > me.x
      say(me, rule.aLine, PAIR_MS)
      say(b, rule.bLine, PAIR_MS)
      b.emoji = rule.bEmoji
      b.right = me.x > b.x
      emit(b)
      clock.clear(b.timer)
      b.timer = running ? clock.set(() => { b.emoji = ''; rest(b) }, PAIR_MS) : null
      doFor(me, 'happy', PAIR_MS, rule.aEmoji)
    })
  }

  // ---------------------------------------------------------------- 开局

  function settle(a) {
    if (away(a)) {
      a.hidden = true
      a.action = 'away'
      return
    }
    const spot = nearFire(a) || randomGrass(a)
    if (spot) {
      a.x = spot.x
      a.y = spot.y
    }
  }

  function walkIn(a) {
    const [ex, ey] = pick(SCENE.entries)
    a.x = ex
    a.y = ey
    say(a, a.pet.lines.hello, HELLO_MS)
    const [bx, by] = SCENE.bowls[0]
    const spot = { x: bx, y: by + 120, action: 'eat' }
    a.goal = { spot, then: (b) => doFor(b, 'eat', EAT_MS, ''), path: detour({ x: a.x, y: a.y }, spot) || [spot] }
  }

  const arriving = new Set(arrivals)
  animals.forEach((a) => (arriving.has(a.id) ? walkIn(a) : settle(a)))

  function stateOf(a) {
    const spot = a.goal && a.goal.spot
    return { action: a.action, decorName: spot && spot.name ? spot.name : '' }
  }

  return {
    sprites: () => animals.map(sprite),

    start() {
      running = true
      animals.forEach((a) => {
        if (a.goal && a.goal.path) return stepAlong(a)
        schedule(a, between(START_MS))
      })
    },

    pause() {
      running = false
      animals.forEach((a) => {
        clock.clear(a.timer)
        a.timer = null
      })
    },

    resume() {
      this.start()
    },

    stop() {
      running = false
      animals.forEach((a) => {
        ;[a.timer, a.bubbleTimer, a.tapTimer].forEach((h) => clock.clear(h))
        a.timer = a.bubbleTimer = a.tapTimer = null
      })
    },

    updateDecor(next) {
      placed = next
      animals.forEach((a) => {
        const uid = a.goal && a.goal.spot && a.goal.spot.uid
        if (uid && !placed.some((p) => p.uid === uid)) {
          a.goal = null
          rest(a)
        }
      })
    },

    tap(i) {
      const a = animals[i]
      if (!a || a.hidden) return
      say(a, lineFor(a.id, { period: periodOf(hour()), action: a.action }, random))
      heart(a)
      emit(a)
    },

    doubleTap(i) {
      const a = animals[i]
      if (!a || a.hidden || a.holdGift) return
      a.goal = a.goal && !a.goal.path ? a.goal : null
      doFor(a, 'roll', ROLL_MS, '😆')
    },

    longPress(i) {
      const a = animals[i]
      if (!a || a.hidden || a.holdGift) return
      a.goal = a.goal && !a.goal.path ? a.goal : null
      doFor(a, 'pet', PET_MS, '❤️')
    },

    feed(i, foodId, liked) {
      const a = animals[i]
      if (!a || a.hidden) return
      a.goal = a.goal && !a.goal.path ? a.goal : null
      say(a, liked ? a.pet.lines.fed : a.pet.lines.dislike)
      doFor(a, liked ? 'eat' : 'dislike', EAT_MS, liked ? '😋' : '💦')
    },

    // 打卡成功：全员跑到篝火边蹦一蹦
    celebrate() {
      animals.forEach((a) => {
        if (a.hidden || a.holdGift) return
        const [sx, sy] = SCENE.seats[a.index % SCENE.seats.length]
        a.goal = null
        if (!walkTo(a, { x: sx, y: sy, action: 'jump' }, (b) => doFor(b, 'jump', JUMP_MS, '🎉'))) doFor(a, 'jump', JUMP_MS, '🎉')
      })
    },

    // 早上的礼物：这只叼着礼物走到画面中间（篝火上方），等你点
    gift(i) {
      const a = animals[i]
      if (!a || a.hidden) return false
      a.holdGift = true
      a.goal = null
      const spot = { x: SCENE.fire[0], y: SCENE.fire[1] + 420, action: 'gift' }
      a.goal = { spot, then: (b) => { b.action = 'gift'; b.emoji = '🎁'; b.duration = 0; emit(b) }, path: detour({ x: a.x, y: a.y }, spot) || [spot] }
      stepAlongGift(a)
      return true
    },

    giftTaken() {
      animals.forEach((a) => {
        if (!a.holdGift) return
        a.holdGift = false
        say(a, '给你的！')
        rest(a)
      })
    },

    whereIs() {
      return animals.map((a) => ({ index: a.index, id: a.id, name: a.pet.name, face: petFace(a.id), x: a.x, y: a.y, action: a.action, text: describe(stateOf(a)) }))
    },

    focus(i) {
      const a = animals[i]
      if (!a) return null
      heart(a)
      emit(a)
      return { x: a.x, y: a.y }
    },

    noSheet(id) {
      bad.add(id)
      animals.forEach((a) => {
        if (a.id === id) emit(a)
      })
    },
  }

  // 叼礼物走路：和普通走路一样，但 holdGift 期间 turn 不接管，所以自己串
  function stepAlongGift(a) {
    const next = a.goal.path.shift()
    a.right = facingFor(next.x - a.x, a.right)
    a.duration = Math.max(300, Math.round((dist(a.x, a.y, next.x, next.y) / a.pet.speed) * 1000))
    a.x = next.x
    a.y = next.y
    a.action = 'gift'
    a.emoji = '🎁'
    emit(a)
    clock.clear(a.timer)
    a.timer = clock.set(() => (a.goal.path.length ? stepAlongGift(a) : a.goal.then(a)), a.duration)
  }
}

module.exports = { createLife }
```

```js
// 让营地自己动起来：篝火火苗和烟、溪流和水塘的波光、涟漪、夜里的萤火虫、水壶炉的热气、花坛的蝴蝶、灯的光晕。全是 CSS 动画，不占包
// 纯函数，有单测；坐标是屏幕像素（画的像素 × L.k）

const { SCENE, scaleAt } = require('./shared/camp')
const { decorById, patchRect } = require('./shared/decor')
const { zOf } = require('./camp-view')

const STEAM_IMAGE = '/images/garden/fx/steam.png'
const FIREFLY_COUNT = 8
const fmt = (n) => Math.round(n * 10) / 10

function noise(n) {
  const x = Math.sin(n * 12.9898 + 78.233) * 43758.5453
  return x - Math.floor(x)
}

// 一个多边形在屏幕上的外框 + 相对外框的 clip-path
function polyBox(poly, L) {
  const xs = poly.map((p) => p[0] * L.k)
  const ys = poly.map((p) => p[1] * L.k)
  const left = Math.min(...xs)
  const top = Math.min(...ys)
  const clip = `polygon(${poly.map((p) => `${fmt(p[0] * L.k - left)}px ${fmt(p[1] * L.k - top)}px`).join(', ')})`
  return { left: fmt(left), top: fmt(top), width: fmt(Math.max(...xs) - left), height: fmt(Math.max(...ys) - top), clip }
}

function at(x, y, w, h, L) {
  return { left: fmt(x * L.k - (w * L.k) / 2), top: fmt(y * L.k - h * L.k), width: fmt(w * L.k), height: fmt(h * L.k) }
}

function fxViews(L, hour, placed) {
  const night = hour >= 19 || hour < 6
  const out = []
  const [fx, fy] = SCENE.fire
  const fireZ = zOf(fy) + 5
  for (let i = 0; i < 3; i++) {
    const w = 260 - i * 50
    out.push({ key: `flame-${i}`, kind: 'flame', ...at(fx + (i - 1) * 30, fy - 20, w, w * 1.3, L), z: fireZ + i, delay: fmt(i * 0.3) })
  }
  for (let i = 0; i < 3; i++) {
    out.push({ key: `smoke-${i}`, kind: 'smoke', ...at(fx + (i - 1) * 40, fy - 240, 200, 200, L), z: fireZ + 4, delay: fmt(i * 1.4) })
  }
  ;[['stream', SCENE.nogo.stream], ['pond', SCENE.nogo.pond]].forEach(([name, poly], i) => {
    out.push({ key: `shimmer-${name}`, kind: 'shimmer', ...polyBox(poly, L), z: 2, delay: fmt(i * 2.1) })
  })
  SCENE.pond.spots.forEach(([x, y], i) => {
    out.push({ key: `ripple-${i}`, kind: 'ripple', ...at(x, y + 30, 260, 90, L), z: 3, delay: fmt(i * 1.1) })
  })
  if (night) {
    for (let i = 0; i < FIREFLY_COUNT; i++) {
      const x = 400 + noise(i * 3 + 1) * (SCENE.width - 800)
      const y = 900 + noise(i * 3 + 2) * (SCENE.height - 1500)
      out.push({ key: `firefly-${i}`, kind: 'firefly', ...at(x, y, 24, 24, L), z: 1900, delay: fmt(noise(i * 3 + 3) * 4) })
    }
  }
  placed.forEach((p) => {
    const d = decorById(p.id)
    if (!d || !d.fx) return
    const rect = patchRect(p.id, p.x, p.y)
    const s = scaleAt(p.y)
    const cx = rect.left + (d.patch.w / 2) * s
    const top = rect.top + d.patch.h * 0.25 * s
    if (d.fx === 'steam') out.push({ key: `steam-${p.uid}`, kind: 'steam', image: STEAM_IMAGE, ...at(cx, top, 220, 220, L), z: zOf(p.y) + 2, delay: 0 })
    if (d.fx === 'butterfly' && !night) out.push({ key: `butterfly-${p.uid}`, kind: 'butterfly', ...at(cx, top, 70, 50, L), z: 1800, delay: fmt(noise(p.x) * 3) })
    if (d.fx === 'glow' && night) out.push({ key: `glow-${p.uid}`, kind: 'glow', ...at(cx, rect.top + d.patch.h * 0.6 * s, d.patch.w * 1.3, d.patch.h * 1.3, L), z: zOf(p.y) + 2, delay: fmt(noise(p.y) * 2) })
  })
  return out
}

module.exports = { fxViews }
```

```js
    if (!d) return
    this.startMove(d)

  queryWorldRect(cb) {
    wx.createSelectorQuery()
      .select('#world')
      .boundingClientRect()
      .exec((res) => {
        if (res && res[0]) this.worldRect = res[0]
        if (cb) cb()
      })

  onDragStart(e) {
    const t = e.touches && e.touches[0]
    if (!t || !this.selected) return
    this.queryWorldRect(() => this.updateDrag(t.clientX, t.clientY))
  },

  onDragMove(e) {
    const t = e.touches && e.touches[0]
    if (!t || !this.selected) return
    this.updateDrag(t.clientX, t.clientY)
  },

  onDragEnd(e) {
    const t = (e.changedTouches && e.changedTouches[0]) || (e.touches && e.touches[0])
    this.setData({ drag: null })

    const p = V.fromTouch(t.clientX, t.clientY, this.worldRect, this.L)
    if (this.selected.type === 'feed') this.feedAt(p)
    else this.placeAt(p)
  },

  updateDrag(x, y) {
    const sel = this.selected
    if (!sel || !this.worldRect) return
    const p = V.fromTouch(x, y, this.worldRect, this.L)
    if (sel.type === 'feed') {
      const size = 56
      this.setData({ drag: { image: V.foodImage(sel.foodId), left: p.x * this.L.k - size / 2, top: p.y * this.L.k - size * 1.6, width: size, height: size, ok: !!this.hitTarget(p), foot: null } })
      return
    }
    const problem = placeProblem(this.camp.decor.placed, sel.decorId, p.x, p.y, sel.type === 'move' ? sel.uid : undefined)
    this.setData({ drag: V.ghostView(sel.decorId, p.x, p.y, this.L, !problem) })
  },

  // 松手时手指在谁身上：宠物（外框）或食盆
  hitTarget(p) {
    const L = this.L
    const px = p.x * L.k
    const py = p.y * L.k
    const pets = this.data.pets
    for (let i = pets.length - 1; i >= 0; i--) {
      const s = pets[i]
      if (s.hidden) continue
      const top = L.height - s.bottom - s.width
      if (px >= s.left && px <= s.left + s.width && py >= top && py <= L.height - s.bottom) return { pet: i }
    }
    for (const b of this.data.bowls) {
      if (px >= b.left && px <= b.left + b.size && py >= b.top && py <= b.top + b.size) return { bowl: b.index }
    }
    return null
  },

  feedAt(p) {
    const sel = this.selected
    const hit = this.hitTarget(p)
    if (!hit) {
      wx.showToast({ title: '拖到小动物或者食盆上松手', icon: 'none' })
      return
    }
    if (hit.bowl !== undefined) {
      this.save('camp.putFood', { bowl: hit.bowl, foodId: sel.foodId })
      return
    }
    const sprite = this.data.pets[hit.pet]
    const index = hit.pet
    api.call('camp.feed', { petId: sprite.id, foodId: sel.foodId })
      .then((view) => {
        this.cancel()
        if (this.life) this.life.feed(index, sel.foodId, view.liked)
        this.render(view)
        wx.showToast({ title: view.liked ? `${sprite.name}吃得很开心，亲密 +${view.gain}` : `${sprite.name}不太爱吃，亲密 +${view.gain}`, icon: 'none' })
      })
      .catch((err) => api.showError(err))
  },

  placeAt(p) {
    const sel = this.selected
    const problem = placeProblem(this.camp.decor.placed, sel.decorId, p.x, p.y, sel.type === 'move' ? sel.uid : undefined)
    if (problem) {
      wx.showToast({ title: problem, icon: 'none' })
      return
    }
    const x = Math.round(p.x)
    const y = Math.round(p.y)
    if (sel.type === 'place') this.save('camp.place', { decorId: sel.decorId, x, y })
    else this.save('camp.move', { uid: sel.uid, x, y })
  },

  save(action, data, toast) {
    if (this.saving) return
    this.saving = true
    api.call(action, data)
      .then((view) => {
        this.saving = false
        this.cancel()
        this.render(view)
        if (toast) wx.showToast({ title: toast, icon: 'none' })
      })
      .catch((err) => {
        this.saving = false
        api.showError(err)
      })
  },

  cancel() {
    this.selected = null
    this.setData({ mode: 'view', selectedName: '', drag: null })
    if (this.life) this.life.resume()
  },

  // ---------------------------------------------------------------- 找它

  openResidents() {

  },

  closeResidents() {
    this.setData({ residentsOpen: false })
  },

  findResident(e) {

    this.setData({ residentsOpen: false })
    if (at) this.nudgeView(V.focusOffset(at.x, at.y, this.L, this.win(), 1.6))
  },

  // ---------------------------------------------------------------- 跳转
```
